# ตัวอย่างการเรียก API ด้วย curl

$ curl -X GET "{{ request.host_url }}api/v1/tickets?status=Pending" \

-H "Cookie: session=YOUR_SESSION_COOKIE"

# สร้าง Ticket ใหม่

$ curl -X POST "{{ request.host_url }}api/v1/tickets" \

-H "Content-Type: application/json" \

-d '{"title": "เครื่องพิมพ์เสีย", "priority": "High", "category_id": 1}'
